{{- /* OG / Twitter / JSON-LD で使う代表画像を 1 枚解決し、dict（url / width / height / type / alt）を返す。 url が空なら画像なし。探す順は Hugo の慣習に合わせる: 1. front matter の images（配列。先頭を使う）/ image（文字列） 2. ページバンドル内の画像で名前に feature / cover / thumbnail を含むもの 3. 事前生成されたページ別 OG 画像（static/images/og/[/].png） 4. site.Params.images（サイト既定。先頭を使う） パスはページバンドル → グローバル assets → static/ 直配置の順で解決し、 実寸（width / height）が取れるときだけ返す（render-image.html と同じ流儀）。 type は og:image:type 用の MIME（リソースなら MediaType、static/ 直配置なら拡張子から）。 alt は og:image:alt / twitter:image:alt 用。front matter の imageAlt → ページ固有の画像 （1〜3）ならページタイトル → site.Params.imageAlt → サイト名の順。X の上限に合わせて 420 文字で切る。 */ -}} {{- $src := "" -}} {{- $res := "" -}} {{- $pageSpecific := false -}} {{- $siteDefault := "" -}} {{- with site.Params.images }}{{ $siteDefault = cond (reflect.IsSlice .) (index . 0) . }}{{ end -}} {{- /* static/ 直配置・外部 URL の og:image:type は拡張子から引く（リソースなら MediaType を使う）。 */ -}} {{- $types := dict ".png" "image/png" ".jpg" "image/jpeg" ".jpeg" "image/jpeg" ".gif" "image/gif" ".webp" "image/webp" ".avif" "image/avif" ".svg" "image/svg+xml" -}} {{- with .Params.images }}{{ $src = cond (reflect.IsSlice .) (index . 0) . }}{{ end -}} {{- if not $src }}{{ with .Params.image }}{{ $src = . }}{{ end }}{{ end -}} {{- if and (not $src) .IsPage }} {{- with (.Resources.ByType "image").GetMatch "{*feature*,*cover*,*thumbnail*}" }}{{ $res = . }}{{ end -}} {{- end -}} {{- if and (not $src) (not $res) .IsPage .Path }} {{- $defaultLang := hugo.Sites.Default.Language.Lang -}} {{- $candidates := slice (printf "/images/og%s.png" .Path) -}} {{- if ne .Language.Lang $defaultLang }} {{- $candidates = slice (printf "/images/og/%s%s.png" .Language.Lang .Path) | append $candidates -}} {{- end -}} {{- range $candidates }} {{- if and (not $src) (fileExists (printf "static%s" .)) }}{{ $src = . }}{{ end -}} {{- end -}} {{- end -}} {{- /* front matter がサイト既定と同じファイルを名指ししただけなら、ページ固有とはみなさない （alt はその画像を説明する site.Params.imageAlt の方が適切）。 */ -}} {{- if or $res (and $src (ne (strings.TrimPrefix "/" $src) (strings.TrimPrefix "/" $siteDefault))) }}{{ $pageSpecific = true }}{{ end -}} {{- if and (not $src) (not $res) }}{{ $src = $siteDefault }}{{ end -}} {{- if and $src (not $res) }} {{- $res = or (.Resources.Get $src) (resources.Get (strings.TrimPrefix "/" $src)) -}} {{- end -}} {{- $out := dict "url" "" "width" 0 "height" 0 "type" "" "alt" "" -}} {{- with $res }} {{- $out = merge $out (dict "url" .Permalink "type" .MediaType.Type) -}} {{- with try (.Width) }}{{ with .Value }}{{ $out = merge $out (dict "width" . "height" $res.Height) }}{{ end }}{{ end -}} {{- else }} {{- with $src }} {{- /* 先頭の / はサイトルート相対の意味で受けるが、absURL に / 始まりを渡すとホスト直下に 解決されて baseURL のサブパス（例: /contour/）が落ちる。剥がして baseURL 相対で解決する。 外部の絶対 URL は absURL がそのまま返す。 */ -}} {{- $out = merge $out (dict "url" (absURL (strings.TrimPrefix "/" .))) -}} {{- /* 拡張子はクエリやフラグメントを除いたパス部分から取る（?w=1200 などが付いた外部 URL 対策）。 */ -}} {{- $u := urls.Parse . -}} {{- with index $types (lower (path.Ext $u.Path)) }}{{ $out = merge $out (dict "type" .) }}{{ end -}} {{- /* 外部 URL でなければ static/ 配下のファイルとして実寸を読む（先頭の / の有無は問わない）。 */ -}} {{- if and (not $u.Scheme) (not (hasPrefix . "//")) }} {{- with try (imageConfig (printf "static/%s" (strings.TrimPrefix "/" .))) }}{{ with .Value }}{{ $out = merge $out (dict "width" .Width "height" .Height) }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $out.url }} {{- /* 空白だけの指定は未指定扱いにして次の候補へ落とす。 */ -}} {{- $alt := (strings.TrimSpace (.Params.imageAlt | default "")) | default (cond $pageSpecific .Title "") | default (strings.TrimSpace (site.Params.imageAlt | default "")) | default site.Title -}} {{- $out = merge $out (dict "alt" (truncate 420 "" $alt)) -}} {{- end -}} {{- return $out -}}
